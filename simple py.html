Math & Comparisons

# Math
print(10 + 5)    # 15
print(10 / 3)    # 3.333
print(10 // 3)   # 3 (floor)
print(10 % 3)    # 1 (remainder)
print(2 ** 3)    # 8

# Compare (returns True/False)
print(10 > 5)    # True
print(10 == 10)  # True (== is compare, = is assign)
Pro Tip: = assigns values. == compares. Beginners mix these up!

If Statements

Loops

# For loop (lists)
fruits = ["apple", "banana", "orange"]
for fruit in fruits:
    print(fruit)

# For loop (numbers)
for i in range(5):  # 0,1,2,3,4
    print(i)

# While loop
count = 0
while count < 3:
    print(count)
    count += 1

⚙️ Functions

def greet(name="Friend"):
    """Greet someone (docstring explains what it does)"""
    return f"Hello, {name}!"

# Call it
print(greet("Maron"))
print(greet())  # Uses default
Rule: Same code 2+ times? Make it a function!

📁 Files & Error Handling

# Write file
with open("notes.txt", "w") as f:
    f.write("Hello from Python!\n")

# Read file
with open("notes.txt") as f:
    content = f.read()
    print(content)

# Safe errors
try:
    number = int(input("Number: "))
    result = 100 / number
except ValueError:
    print("Not a number!")
except ZeroDivisionError:
    print("Can't divide by 0!")

🏗️ OOP (Classes)

class Dog:
    def __init__(self, name, breed):
        self.name = name
        self.breed = breed

    def bark(self):
        return f"{self.name} says Woof!"

# Use it
my_dog = Dog("Rex", "Labrador")
print(my_dog.bark())
